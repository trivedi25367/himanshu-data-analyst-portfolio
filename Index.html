-- Advanced SQL: Aggregation & Window Function Analysis
WITH MonthlyAttendance AS (
    SELECT 
        student_id,
        DATE_TRUNC('month', attendance_date) AS month_record,
        COUNT(CASE WHEN status = 'Present' THEN 1 END) AS present_days,
        COUNT(attendance_id) AS total_days
    FROM student_attendance
    GROUP BY 1, 2
)
SELECT 
    student_id,
    month_record,
    ROUND((present_days * 100.0 / total_days), 2) AS attendance_percentage,
    DENSE_RANK() OVER(
        PARTITION BY month_record 
        ORDER BY (present_days * 1.0 / total_days) DESC
    ) AS batch_rank
FROM MonthlyAttendance;
# Data Cleaning & Missing Value Imputation with Pandas
import pandas as pd
import numpy as np

def clean_and_aggregate(file_path):
    df = pd.read_csv(file_path)
    # Handle null records and typecasting
    df['date'] = pd.to_datetime(df['date'])
    df['score'] = df['score'].fillna(df.groupby('category')['score'].transform('median'))
    
    # KPI Aggregation
    summary = df.groupby('category').agg(
        total_records=('id', 'count'),
        avg_score=('score', 'mean')
    ).reset_index()
    return summary
